ROOT_RESOURCE_ID=$(aws apigateway get-resources \
  --rest-api-id "$API_ID" \
  --query "items[?path=='/'].id | [0]" \
  --output text \
  --endpoint-url http://localhost:4566 \
  --region us-east-1)

TOKEN_RESOURCE_ID=$(aws apigateway create-resource \
  --rest-api-id "$API_ID" \
  --parent-id "$ROOT_RESOURCE_ID" \
  --path-part token \
  --query id \
  --output text \
  --endpoint-url http://localhost:4566 \
  --region us-east-1)

COUPONS_RESOURCE_ID=$(aws apigateway get-resources \
  --rest-api-id "$API_ID" \
  --query "items[?path=='/coupons'].id | [0]" \
  --output text \
  --endpoint-url http://localhost:4566 \
  --region us-east-1)

TO_SECURE_RESOURCE_ID=$(aws apigateway create-resource \
  --rest-api-id "$API_ID" \
  --parent-id "$COUPONS_RESOURCE_ID" \
  --path-part to_secure \
  --query id \
  --output text \
  --endpoint-url http://localhost:4566 \
  --region us-east-1)
